-- 3-1) TEST_EMP 테이블에서 사원번호가 7369인 사원의 급여를 3000으로 수정
UPDATE TEST_EMP
SET sal = 3000
WHERE empno = 7369;

SELECT * 
FROM TEST_EMP
WHERE empno = 7369;

-- 4. DEPTNO가 30인 사원들의 급여를 모두 10% 인상
UPDATE TEST_EMP
SET comm = comm * 1.1
WHERE deptno = 30;

SELECT deptno, ename, comm
FROM TEST_EMP
ORDER BY deptno DESC;

-- 4-1) TEST_EMP 테이블에서 커미션이 NULL인 사원 중에서 급여가 1000 미만인 사원을 삭제
DELETE FROM TEST_EMP
WHERE comm IS NULL AND sal < 1000;

SELECT ename, sal, comm
FROM test_emp
WHERE comm IS NULL
ORDER BY sal ASC;
